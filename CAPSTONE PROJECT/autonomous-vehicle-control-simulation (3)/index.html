MODULE 1 OOP VEHICLE CONTROL

Autonomous Vehicle Control Using OOP (Base Class Vehicle)

Implements core vehicle drivetrain operations using Object-Oriented Programming concepts: encapsulation of speed, direction, and status, inheritance hierarchy, and runtime polymorphism via virtual functions.

OOP CONCEPTS & BASE CLASS DESIGN
1. ENCAPSULATION (Protected Data Members)
  • protected: double speed; — Validated within 0.0 – 120.0 km/h.
  • protected: string direction; — Validated movement vector (FORWARD, BACKWARD, LEFT, RIGHT).
  • protected: string status; — Operating state (STOPPED, MOVING, EMERGENCY_STOP).
2. RUNTIME POLYMORPHISM (Virtual Member Functions)
  • virtual bool start(); / virtual bool stop();
  • virtual bool accelerate(double delta = 10.0);
  • virtual bool decelerate(double delta = 10.0);
  • virtual bool setDirection(const string& newDir);
  • virtual void displayStatus() const; (Overridden by derived classes)

C++ MODULE 1 CODE (Base Class Vehicle) 
class Vehicle {
protected:
    double speed;
    string direction;
    string status;

public:
    Vehicle(string initialDirection = "FORWARD")
        : speed(0.0), direction(initialDirection), status("STOPPED") {}
    virtual ~Vehicle() {}

    double getSpeed() const { return speed; }
    string getDirection() const { return direction; }
    string getStatus() const { return status; }

    virtual bool start() {
        status = "STOPPED";
        cout << "Vehicle started successfully." << endl;
        return true;
    }

    virtual bool stop() {
        speed = 0.0;
        status = "STOPPED";
        cout << "Vehicle stopped successfully." << endl;
        return true;
    }

    virtual bool accelerate(double delta = 10.0) {
        speed = min(120.0, speed + delta);
        status = (speed > 0.0) ? "MOVING" : "STOPPED";
        return true;
    }

    virtual bool decelerate(double delta = 10.0) {
        if (speed <= 0.0) return false;
        speed = max(0.0, speed - delta);
        if (speed == 0.0) status = "STOPPED";
        return true;
    }

    virtual bool setDirection(const string& newDir) {
        if (newDir != "FORWARD" && newDir != "BACKWARD" &&
            newDir != "LEFT" && newDir != "RIGHT") return false;
        direction = newDir;
        return true;
    }

    virtual void displayStatus() const {
        cout << "Status: " << status
             << " | Speed: " << speed << " km/h"
             << " | Direction: " << direction << endl;
    }
};
MODULE 2 NAVIGATION & OBSTACLE DETECTION

Navigation, Obstacle Proximity Radar & Danger Zone Solutions

Extends Vehicle via inheritance (class AutonomousCar : public Vehicle) to enforce safety thresholds, freeze motion in danger zones (distance ≤ safetyThreshold), and compute automated recovery solutions.

NAVIGATION RULES, THRESHOLD & DANGER SOLUTIONS
1. DERIVED CLASS: AutonomousCar : public Vehicle
  • Multi-Lane Obstacle Radar: Tracks struct Obstacle { id, type, lane, distance } across LEFT, CENTER, and RIGHT lanes.
  • Safety Threshold Interlock: If distance ≤ safetyThreshold (default 5.0m), vehicle stops immediately (speed = 0, EMERGENCY_STOP) and locks movement.
  • Danger Zone Solutions: Evaluates side lanes for BYPASS_LEFT or BYPASS_RIGHT; falls back to REVERSE_BACKOUT (+10m) or CLEAR_OBSTACLE when all lanes are blocked.
SAFETY THRESHOLD & DANGER SOLUTION TRIGGERS:
STRICT IMMOBILIZATION RULE

While the 2D Simulated Road is in danger, acceleration and forward navigation remain strictly disabled until a Danger Zone Solution moves the vehicle to safety.

C++ MODULE 2 CODE (Derived Class AutonomousCar) 
struct Obstacle {
    string id, type, lane;
    double distance;
};

class AutonomousCar : public Vehicle {
private:
    double safetyThreshold, lastObstacleDist;
    bool emergencyBrakeActive;
    string currentLane;
    vector<Obstacle> activeObstacles;

public:
    AutonomousCar(double threshold = 5.0)
        : Vehicle("FORWARD"), safetyThreshold(threshold),
          lastObstacleDist(100.0), emergencyBrakeActive(false), currentLane("CENTER") {}

    bool detectObstacle(double distance, const string& type = "Road Barricade") {
        if (distance < 0.0) return false;
        lastObstacleDist = distance;
        activeObstacles.push_back({"OBS-" + to_string(activeObstacles.size() + 1), type, currentLane, distance});

        if (distance <= safetyThreshold) {
            emergencyBrakeActive = true;
            speed = 0.0;
            status = "EMERGENCY_STOP";
            displayActiveObstacles();
            displayDangerZoneSolution(); // BYPASS_LEFT / BYPASS_RIGHT / REVERSE_BACKOUT
            return false;
        }
        emergencyBrakeActive = false;
        return true;
    }

    bool resolveDangerZone(const string& solutionType = "AUTO") {
        string chosen = (solutionType == "AUTO") ? getRecommendedDangerSolution() : solutionType;
        if (chosen == "BYPASS_LEFT") { currentLane = "LEFT"; lastObstacleDist = 65.0; }
        else if (chosen == "BYPASS_RIGHT") { currentLane = "RIGHT"; lastObstacleDist = 65.0; }
        else if (chosen == "REVERSE_BACKOUT") { lastObstacleDist += 10.0; }
        else { lastObstacleDist = 75.0; }
        emergencyBrakeActive = (lastObstacleDist <= safetyThreshold);
        status = emergencyBrakeActive ? "EMERGENCY_STOP" : "STOPPED";
        return !emergencyBrakeActive;
    }
};

#include <iostream>
#include <string>
#include <vector>
#include <iomanip>
#include <algorithm>

using namespace std;

const double DEFAULT_SAFETY_THRESHOLD = 5.0; // meters
const double MAX_SPEED = 120.0;              // km/h
const double SPEED_STEP = 10.0;

struct Obstacle {
    string id, type, lane;
    double distance;
};

// MODULE 1: Autonomous Vehicle Control Using OOP (Base Class Vehicle)
class Vehicle {
protected:
    double speed;
    string direction, status;
public:
    Vehicle(string initialDirection = "FORWARD")
        : speed(0.0), direction(initialDirection), status("STOPPED") {}
    virtual ~Vehicle() {}

    virtual bool start() { status = "STOPPED"; return true; }
    virtual bool stop() { speed = 0.0; status = "STOPPED"; return true; }
    virtual bool accelerate(double delta = SPEED_STEP) {
        speed = min(MAX_SPEED, speed + delta);
        status = (speed > 0) ? "MOVING" : "STOPPED";
        return true;
    }
    virtual bool decelerate(double delta = SPEED_STEP) {
        if (speed <= 0.0) return false;
        speed = max(0.0, speed - delta);
        if (speed == 0.0) status = "STOPPED";
        return true;
    }
    virtual bool setDirection(const string& newDir) {
        if (newDir != "FORWARD" && newDir != "BACKWARD" && newDir != "LEFT" && newDir != "RIGHT") return false;
        direction = newDir;
        return true;
    }
    virtual void displayStatus() const {
        cout << "Status: " << status << " | Speed: " << speed << " km/h\n";
    }
};

// MODULE 2: Navigation & Obstacle Detection (Derived Class AutonomousCar)
class AutonomousCar : public Vehicle {
private:
    double safetyThreshold, lastObstacleDist;
    bool emergencyBrakeActive;
    string currentLane;
    vector<Obstacle> activeObstacles;
public:
    AutonomousCar(double threshold = DEFAULT_SAFETY_THRESHOLD)
        : Vehicle("FORWARD"), safetyThreshold(threshold),
          lastObstacleDist(100.0), emergencyBrakeActive(false), currentLane("CENTER") {}

    void displayActiveObstacles() const {
        cout << "\n[DETECTED OBSTACLES RADAR]\n";
        for (const auto& obs : activeObstacles) {
            string sev = (obs.distance <= safetyThreshold) ? "DANGER ZONE" : "SAFE";
            cout << " - " << obs.id << " | " << obs.type << " | Lane: " << obs.lane
                 << " | " << obs.distance << "m [" << sev << "]\n";
        }
    }

    string getRecommendedDangerSolution() const {
        if (lastObstacleDist > safetyThreshold) return "PATH_SAFE";
        auto laneClear = [&](const string& l) {
            for (const auto& o : activeObstacles) if (o.lane == l && o.distance <= safetyThreshold) return false;
            return true;
        };
        if (currentLane == "CENTER") {
            if (laneClear("LEFT")) return "BYPASS_LEFT";
            if (laneClear("RIGHT")) return "BYPASS_RIGHT";
        }
        return "REVERSE_BACKOUT";
    }

    bool resolveDangerZone(const string& sol = "AUTO") {
        string chosen = (sol == "AUTO") ? getRecommendedDangerSolution() : sol;
        if (chosen == "BYPASS_LEFT") { currentLane = "LEFT"; lastObstacleDist = 65.0; }
        else if (chosen == "BYPASS_RIGHT") { currentLane = "RIGHT"; lastObstacleDist = 65.0; }
        else if (chosen == "REVERSE_BACKOUT") { lastObstacleDist += 10.0; }
        else { lastObstacleDist = 75.0; }
        emergencyBrakeActive = (lastObstacleDist <= safetyThreshold);
        status = emergencyBrakeActive ? "EMERGENCY_STOP" : "STOPPED";
        cout << "[Danger Solution] Applied " << chosen << ". New clearance: " << lastObstacleDist << "m.\n";
        return !emergencyBrakeActive;
    }

    bool detectObstacle(double distance, const string& type = "Road Barricade") {
        if (distance < 0.0) return false;
        lastObstacleDist = distance;
        activeObstacles.push_back({"OBS-" + to_string(activeObstacles.size() + 1), type, currentLane, distance});
        if (distance <= safetyThreshold) {
            emergencyBrakeActive = true; speed = 0.0; status = "EMERGENCY_STOP";
            cout << "Output: Obstacle detected!\nOutput: Emergency stop activated.\n";
            displayActiveObstacles();
            cout << "Recommended Solution: " << getRecommendedDangerSolution() << "\n";
            return false;
        }
        emergencyBrakeActive = false;
        cout << "Output: Path safe.\n";
        return true;
    }
};
